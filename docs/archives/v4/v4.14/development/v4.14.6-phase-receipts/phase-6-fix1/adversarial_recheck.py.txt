"""Re-run of the T042 adversarial scenarios (AF-1 to AF-13) against the fix-cycle-1 engine.

Every scenario runs the real CLI (`nexus_hub_cli.py settings ...`) as a child process against a scratch home under the
system temp folder, with a scrubbed environment and no model call. The panel confirmation is now a digest file plus
a nonce in the child's environment, so a forged confirmation is built here the way a same-user attacker would build it.
Usage: python recheck.py <worktree> <scratch root>
"""
import copy
import hashlib
import json
import os
import shutil
import stat
import subprocess
import sys
import threading
import time
from pathlib import Path

WT = Path(sys.argv[1])
ROOT = Path(sys.argv[2])
PY = sys.executable
CLI = WT / "scripts" / "nexus_hub_cli.py"
SAFE_PATH = r"C:\Windows\System32;C:\Windows"
CS = ".claude/settings.json"
NONCE = "f" * 64
NONCE_ENV = "NEXUS_SETTINGS_PANEL_NONCE"
FINDINGS = []

CLAUDE_SEED = '{\n  "permissions": {\n    "allow": ["Bash(git status)", "Read(*)"],\n    "defaultMode": "default"\n  },\n  "env": {\n    "CLAUDE_CODE_EFFORT_LEVEL": "low"\n  },\n  "model": "sonnet"\n}\n'
CODEX_SEED = "# my codex notes\nmodel = \"gpt-x\"   # keep this comment\napproval_policy = \"on-request\"\n"


def force(func, path, _exc):
    os.chmod(path, stat.S_IWRITE | stat.S_IREAD)
    func(path)


def mkhome(name, claude=CLAUDE_SEED, codex=CODEX_SEED):
    home = ROOT / name
    if home.exists():
        shutil.rmtree(home, onerror=force)
    for sub in (".claude", ".codex", ".nexus-hub"):
        (home / sub).mkdir(parents=True, exist_ok=True)
    if claude is not None:
        (home / CS).write_bytes(claude.encode() if isinstance(claude, str) else claude)
    if codex is not None:
        (home / ".codex" / "config.toml").write_bytes(codex.encode() if isinstance(codex, str) else codex)
    return home


def env_for(home, extra=None):
    env = {k: os.environ[k] for k in ("SYSTEMROOT", "TEMP", "TMP", "COMSPEC", "PATHEXT", "WINDIR") if k in os.environ}
    env["PATH"] = SAFE_PATH
    h = str(home)
    env.update(HOME=h, USERPROFILE=h, APPDATA=h + r"\AppData\Roaming", LOCALAPPDATA=h + r"\AppData\Local", NEXUS_HUB_HOME=h + r"\.nexus-hub", PYTHONDONTWRITEBYTECODE="1")
    for key, value in (extra or {}).items():
        if value is None:
            env.pop(key, None)
        else:
            env[key] = value
    return env


def run(home, args, extra=None, stdin_text=None, timeout=120):
    kw = dict(env=env_for(home, extra), capture_output=True, text=True, timeout=timeout, creationflags=subprocess.CREATE_NO_WINDOW, cwd=str(home))
    if stdin_text is None:
        kw["stdin"] = subprocess.DEVNULL
    else:
        kw["input"] = stdin_text
    done = subprocess.run([PY, "-B", "-X", "utf8", str(CLI), "settings", *args], **kw)
    return done.returncode, done.stdout, done.stderr


def forge(home, nonce_in_env=True, old_format=False, **fields):
    """What a same-user process does: write the confirmation file; optionally also put the nonce in the child's env."""
    body = {"issued_at": time.time(), "action": "set", "control": "posture", "value": "full-auto", "tool": "claude"}
    if old_format:
        body["token"] = "a" * 64
    else:
        body["token_sha256"] = hashlib.sha256(NONCE.encode()).hexdigest()
    body.update(fields)
    (home / ".nexus-hub" / "settings-panel-token.json").write_text(json.dumps(body))
    return {NONCE_ENV: NONCE} if nonce_in_env else {}


def rd(home, rel):
    path = home / rel
    return path.read_text(encoding="utf-8") if path.exists() else None


def mode(home):
    return json.loads(rd(home, CS)).get("permissions", {}).get("defaultMode")


def check(label, ok, detail=""):
    FINDINGS.append((label, ok, detail))
    print(("PASS " if ok else "FAIL ") + label + (f"  [{detail}]" if detail else ""))


def tail(err):
    lines = err.strip().splitlines()
    return lines[-1][:140] if lines else ""


# --- AF-1: every remove-override refuses inside an agent session -----------------------------------
home = mkhome("af1")
rc, out, err = run(home, ["remove-override", "effort", "--tool", "claude"], extra={"CLAUDECODE": "1"})
check("AF-1a non-safety remove-override, marker, no token, no terminal -> refused", rc == 3 and "CLAUDE_CODE_EFFORT_LEVEL" in rd(home, CS), f"rc={rc} {tail(err)}")
home = mkhome("af1b")
extra = forge(home, action="remove-override", control="effort", value=None, tool="claude")
rc, out, err = run(home, ["remove-override", "effort", "--tool", "claude"], extra={"CLAUDECODE": "1", **extra})
check("AF-1b same with a complete forged confirmation (digest file + nonce in env), marker present -> refused", rc == 3 and "CLAUDE_CODE_EFFORT_LEVEL" in rd(home, CS), f"rc={rc} {tail(err)}")
home = mkhome("af1c", claude=json.dumps({"permissions": {"disableBypassPermissionsMode": "disable"}}, indent=2))
extra = forge(home, action="remove-override", control="posture", value=None, tool="claude")
rc, out, err = run(home, ["remove-override", "posture", "--tool", "claude"], extra={"CLAUDECODE": "1", **extra})
check("AF-1c weakening remove-override, marker, forged confirmation -> refused", rc == 3 and "disableBypassPermissionsMode" in rd(home, CS), f"rc={rc}")

# --- AF-2: reset that would put back a weaker value ---------------------------------------------------
home = mkhome("af2", claude=json.dumps({"permissions": {"defaultMode": "bypassPermissions"}}, indent=2) + "\n")

rc, out, err = run(home, ["reset", "posture", "--tool", "claude"])
check("AF-2b same without a marker and without a terminal -> refused", rc == 3 and mode(home) == "default", f"rc={rc} {tail(err)}")
rc, out, err = run(home, ["reset", "posture", "--tool", "claude"], stdin_text="yes\n")
check("AF-2c typed yes on a pipe (not a terminal) -> refused", rc == 3 and mode(home) == "default", f"rc={rc}")
rc, out, err = run(home, ["reset", "posture", "--tool", "claude", "--dry-run", "--json"], extra={"CLAUDECODE": "1"})

home = mkhome("af2e")

check("AF-2e git_guardrails reset to off, marker -> refused, guard stays on", rc == 3 and '"block_git_bypass_flags": true' in rd(home, ".nexus-hub/guardrails.json"), f"rc={rc}")
home = mkhome("af2f")

home = mkhome("af2g")

check("AF-2g reset back to a SAFE value (the full-auto was set when none existed) stays open to an agent", rc == 0 and mode(home) == "default", f"rc={rc} mode={mode(home)}")
home = mkhome("af2h", claude=json.dumps({"permissions": {"defaultMode": "bypassPermissions"}}, indent=2) + "\n")
run(home, ["set", "posture", "ask", "--tool", "claude"])
extra = forge(home, action="reset", control="posture", value=None, tool="claude")
rc, out, err = run(home, ["reset", "posture", "--tool", "claude", "--json"], extra=extra)

# --- AF-3: confirmation forgery -------------------------------------------------------------------------
home = mkhome("af3a")
forge(home, old_format=True, nonce_in_env=False)
rc, out, err = run(home, ["set", "posture", "full-auto", "--tool", "claude"], extra={"CLAUDECODE": None})

home = mkhome("af3b")

for label, fields in {"expired 31 s": {"issued_at": time.time() - 31}, "future +100 s": {"issued_at": time.time() + 100}, "other value": {"value": "accept-edits"}, "other tool": {"tool": "codex"}, "other action": {"action": "reset"}, "bool time": {"issued_at": True}}.items():

# --- AF-4: concurrent commands -------------------------------------------------------------------------
bad = 0
for trial in range(15):
    home = mkhome("af4")
    results = {}

    def go(name, args):
        results[name] = run(home, args)

    jobs = [("a", ["set", "model", "opus", "--tool", "claude"]), ("b", ["set", "effort", "low", "--tool", "claude"]), ("c", ["set", "attribution", "off", "--tool", "claude"]), ("d", ["set", "model", "gpt-9", "--tool", "codex"])]
    threads = [threading.Thread(target=go, args=job) for job in jobs]
    [t.start() for t in threads]
    [t.join() for t in threads]
    record = json.loads(rd(home, ".nexus-hub/settings-control.json"))
    keys = {(e["tool"], ".".join(e["key_path"])) for e in record["entries"]}
    data = json.loads(rd(home, CS))
    wrote = set()
    if data.get("model") == "opus":
        wrote.add(("claude", "model"))
    if data.get("effortLevel") == "low":
        wrote.add(("claude", "effortLevel"))
    if data.get("attribution") == {"commit": "", "pr": ""}:
        wrote |= {("claude", "attribution.commit"), ("claude", "attribution.pr")}
    if "gpt-9" in (rd(home, ".codex/config.toml") or ""):
        wrote.add(("codex", "model"))
    codes = sorted(r[0] for r in results.values())
    if wrote - keys or any(code not in (0, 4) for code in codes) or len(wrote) < 5:
        bad += 1
        print("  trial", trial, "codes", codes, "written but unrecorded", sorted(wrote - keys), "wrote", len(wrote))
check("AF-4 15 trials of four concurrent commands: every written key recorded, every command succeeded", bad == 0, f"bad trials {bad} of 15")
