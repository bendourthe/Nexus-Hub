### Recurrence Qualification

This bounded automated check tests the delivered document's structure and explicit branches. It rejects a missing clause, an unconditional trigger, duplicated ownership, a missing baseline-unavailable branch, relocation outside the existing owner, a removed safeguard, and project-specific baseline constants. It cannot prove that an agent will make a sound necessity judgment. It is not a new shipped validator or a persistent test source.

Run on Windows in PowerShell 5.1 from this worktree. Expected result: eight passing tests and temporary files removed. No provider, network, repository mutation, or user-home instruction write occurs.

```powershell
$probeDir = Join-Path ([System.IO.Path]::GetTempPath()) ('nexus-instruction-owner-' + [guid]::NewGuid().ToString('N'))
New-Item -ItemType Directory -Path $probeDir | Out-Null
$probeFile = Join-Path $probeDir 'test_instruction_owner.py'
$probeCode = @'
from pathlib import Path
import os
import re
import pytest

ROOT = Path(os.environ['NEXUS_INSTRUCTION_OWNER_PROBE_ROOT'])
TEXT = (ROOT / 'catalog/commands/update.md').read_text(encoding='utf-8')
MARKER = 'When this release changes a project instruction file or substantive instruction templates'

def clause(text):
    matches = re.findall(r'^    (' + re.escape(MARKER) + r'[^\n]+)$', text, re.M)
    assert len(matches) == 1, 'one conditional instruction-review handoff required'
    return matches[0]

def qualify(text):
    value = clause(text)
    owners = re.findall(r'^2\. \*\*Full architecture refactor\*\*.*?(?=^2a\. )', text, re.M | re.S)
    assert len(owners) == 1 and value in owners[0], 'handoff must stay in its existing owner'
    assert 'review only the changed standing rules' in value
    assert 'last available local instruction audit' in value
    assert 'if no baseline is available, say so explicitly' in value
    assert 'Do not run a full instruction audit for an unrelated release' in value
    assert not re.search(r'\b(?:every|each) release\b', value, re.I)
    assert all(word in value for word in ('keep', 'make contextual', 'remove', 'defer'))
    assert all(word in value for word in ('trust-boundary', 'data-loss', 'security', 'accessibility', 'verification'))
    assert not re.search(r'\b(?:165|572|16243)\b|v4\.15|instruction-necessity-audit\.md', value)
    return value

def test_current_conditional_owner_and_both_baseline_branches():
    value = qualify(TEXT)
    assert 'before-and-after line counts and approximate token counts' in value

@pytest.mark.parametrize('mutation', ['missing', 'unconditional', 'duplicate', 'no-baseline', 'relocated', 'no-safeguard', 'project-constants'])
def test_independent_negative_controls(mutation):
    value = clause(TEXT)
    if mutation == 'missing':
        changed = TEXT.replace('    ' + value, '', 1)
    elif mutation == 'unconditional':
        changed = TEXT.replace(MARKER, 'For every release', 1)
    elif mutation == 'duplicate':
        changed = TEXT + '\n    ' + value + '\n'
    elif mutation == 'no-baseline':
        changed = TEXT.replace('if no baseline is available, say so explicitly', 'assume the baseline exists', 1)
    elif mutation == 'relocated':
        changed = TEXT.replace('    ' + value, '', 1) + '\n99. **New instruction audit owner**\n\n    ' + value + '\n'
    elif mutation == 'no-safeguard':
        changed = TEXT.replace('accessibility, and verification safeguards', 'accessibility safeguards', 1)
    else:
        changed = TEXT.replace(value, value + ' Use 165 rules and 572 lines.', 1)
    assert changed != TEXT
    with pytest.raises(AssertionError):
        qualify(changed)
'@
```
